cmake_minimum_required(VERSION 3.16)
project(pilimini_player LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PLAYER_LIB "${CMAKE_CURRENT_SOURCE_DIR}/../../../libs/${OHOS_ARCH}/libmpv.so")
if(NOT EXISTS "${PLAYER_LIB}")
  message(FATAL_ERROR "Missing ${OHOS_ARCH} libmpv. Run node tools/prepare-player-deps.cjs and prepare the matching simulator library.")
endif()
add_library(mpv SHARED IMPORTED)
set_target_properties(mpv PROPERTIES IMPORTED_LOCATION "${PLAYER_LIB}")
add_library(pilimini_player SHARED mpv_bridge.cpp)
target_include_directories(pilimini_player PRIVATE third_party)
target_link_libraries(pilimini_player PRIVATE mpv libace_napi.z.so libhilog_ndk.z.so)
target_compile_options(pilimini_player PRIVATE -Wall -Wextra -Werror=return-type)
